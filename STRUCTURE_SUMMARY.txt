🚀 QUICK START COMMANDS

# Install all dependencies
pnpm install --recursive

# Start development (both services)
pnpm dev

# Or start individually
pnpm backend:dev              # Terminal 1
pnpm frontend:dev             # Terminal 2

# Build for production
pnpm build

# Docker deployment
docker-compose up -d
docker-compose down
